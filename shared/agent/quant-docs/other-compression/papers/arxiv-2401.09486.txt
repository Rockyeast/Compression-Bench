 1   class LoMAGenerator:
 2       def __init__(self, model, max_len, position_type, compress_ratio, mem_len,
            mem_token_id) -> None:
 3           self.model = model
 4           self.model_dtype = model.parameters().__next__().dtype
 5           self.max_len = max_len
 6           self.mem_len = mem_len
 7           self.compress_ratio = compress_ratio
 8           self.position_type = position_type
 9           self.read_len = compress_ratio * mem_len
10           self.mem_token_id = mem_token_id
11           self.reset()
12           self.position_ids = None
13
14       def reset(self):
15           self.cursor = 0
16           self.kv_cache = None
17           self.generated = []
18           self.compressed_chunks = 0
19           self.input_buffer = []
20           self.position_ids = None
21
22       def __call__(self, input_ids, eos_token_id):
23           generated = []
24           generated.append(self.add_token_ids(input_ids))
25           while len(generated) < self.max_len and generated[-1] != eos_token_id:
26               generated.append(self.add_token_ids([generated[-1]]))
27           return generated
28
29       def add_token_ids(self, token_ids):
30           assert len(token_ids) > 0
31           self.input_buffer.extend(token_ids)
32           while self.mem_len > 0:
33               # WHILE loop iterates inference and compression
34               kv_cache_len = self.kv_cache[0][0].shape[2] if self.kv_cache else 0
35               uncomp_cache_len = kv_cache_len - self.compressed_chunks * self.
                    mem_len
36               assert uncomp_cache_len < self.read_len
37               proc_len = self.read_len - uncomp_cache_len
38               if len(self.input_buffer) < proc_len:
39                   break
40               input_ids = self.input_buffer[:proc_len]
41               self.input_buffer = self.input_buffer[proc_len:]
42               last_predict = self.inference_input(input_ids)
43               self.compress_last_chunk()
44           if self.input_buffer:
45               last_predict = self.inference_input(self.input_buffer)
46               self.input_buffer = []
47           return last_predict
48

50           input_ids = torch.LongTensor(input_ids)
